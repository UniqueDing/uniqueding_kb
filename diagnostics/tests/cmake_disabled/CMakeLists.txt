cmake_minimum_required(VERSION 3.20)
project(charybdis_diag_disabled_fixture C)

get_filename_component(DIAG_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(CONFIG_CHARYBDIS_TRACKBALL_DIAGNOSTICS OFF)

add_executable(zephyr_final main.c)
add_library(app STATIC main.c)
add_subdirectory("${DIAG_REPO_ROOT}/diagnostics" diagnostics_module)

get_target_property(app_sources app SOURCES)
if("${app_sources}" MATCHES "trackball_diag|diag_stats")
  message(FATAL_ERROR "diagnostics-disabled app unexpectedly acquired sources")
endif()
get_target_property(app_definitions app COMPILE_DEFINITIONS)
if("${app_definitions}" MATCHES "CHARYBDIS_DIAG")
  message(FATAL_ERROR "diagnostics-disabled app unexpectedly acquired compile definitions")
endif()
get_target_property(final_link_options zephyr_final LINK_OPTIONS)
if("${final_link_options}" MATCHES "--wrap=(input_report|zmk_hog_send_mouse_report)")
  message(FATAL_ERROR "diagnostics-disabled target unexpectedly acquired linker wrapping")
endif()
get_property(post_build_set GLOBAL PROPERTY extra_post_build_commands SET)
if(post_build_set)
  get_property(post_build GLOBAL PROPERTY extra_post_build_commands)
  if("${post_build}" MATCHES "verify_postlink.py|TARGET_FILE:zephyr_final")
    message(FATAL_ERROR "diagnostics-disabled target unexpectedly acquired a post-build command")
  endif()
endif()
